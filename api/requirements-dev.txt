# Host-side test dependencies (decision D-1 in specs/001-schema-tool-plan.md).
# Deliberately NOT installed into the runtime image.
#
# SQLAlchemy is pinned to the same range as api/requirements.txt on purpose:
# AC5 asserts exact rendered type strings, so a version skew between the host
# suite and the container would surface as a confusing assertion failure rather
# than as the version problem it actually is.
pytest>=8.0,<9.0
sqlalchemy>=2.0,<2.1
psycopg[binary]>=3.2,<3.3

# The eval dataset is YAML (specs/006-evals.md §5). Dev-only on purpose: evals/
# is a measurement harness and is not shipped in the API image, so adding a
# parser to the runtime image for a benchmark that never runs there would be
# dead weight in production.
pyyaml>=6.0,<7.0

# Token measurement (specs/008-prompt-tuning.md D-1). Dev-only, and for the
# same reason as pyyaml: this is measurement apparatus, not runtime code, and
# shipping a tokenizer in the API image would be dead weight in production.
#
# o200k_base is the encoding of the deployed openai/gpt-oss-* family. It is a
# tokenizer, not a provider SDK, so it does not engage the section 5
# swappability constraint -- that guards `groq` imports under api/.
#
# It exists because T1 found the spec's headline token numbers were len//4.
tiktoken>=0.7,<1.0
